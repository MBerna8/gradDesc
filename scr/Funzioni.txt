#include <cmath>

const double PI = std::acos(-1.0);

// Funzione 1
double Function1::f(double x, double y){
    return (x*x + y*y)/2;
}
double* Function1::grad(double x, double y){
    return new double[2]{x, y};
}

// Funzione 2
double Function2::f(double x, double y){
    return (x*x + 16*y*y)/2;
}
double* Function2::grad(double x, double y){
    return new double[2]{x, 16*y};
}

// Funzione 3
double Function3::f(double x, double y){
    return (x*x - 1)*(x*x - 1) + y*y;
}
double* Function3::grad(double x, double y){
    return new double[2]{4*x*(x*x - 1), 2*y};
}

// Funzione 4
double Function4::f(double x, double y){
    return (x*x - 1)*(x*x - 1) + (x - 1)*(x - 1)/5 + y*y;
}
double* Function4::grad(double x, double y){
    return new double[2]{4*x*(x*x - 1) + 2*(x - 1)/5, 2*y};
}

// Funzione 5
double Function5::f(double x, double y){
    double s = std::sinh(x/2);
    return 4*s*s + y*y;
}
double* Function5::grad(double x, double y){
    return new double[2]{2*std::sinh(x), 2*y};
}

// Funzione 6
double Function6::f(double x, double y){
    return std::log(1 + x*x + y*y);
}
double* Function6::grad(double x, double y){
    double d = 1 + x*x + y*y;
    return new double[2]{2*x/d, 2*y/d};
}

// Funzione 7
double Function7::f(double x, double y){
    return 1 - std::exp(-(x*x + y*y));
}
double* Function7::grad(double x, double y){
    double e = std::exp(-(x*x + y*y));
    return new double[2]{2*x*e, 2*y*e};
}

// Funzione 8
double Function8::f(double x, double y){
    return 2 - std::cos(PI*x) - std::cos(PI*y);
}
double* Function8::grad(double x, double y){
    return new double[2]{PI*std::sin(PI*x), PI*std::sin(PI*y)};
}
