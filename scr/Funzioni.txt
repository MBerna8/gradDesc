// Funzione 1
double Function1::f(double x, double y){
    return (x*x + y*y)/2;
}
std::array<double,2> Function1::grad(double x, double y){
    return {x, y};
}

// Funzione 2
double Function2::f(double x, double y){
    return (x*x + 16*y*y)/2;
}
std::array<double,2> Function2::grad(double x, double y){
    return {x, 16*y};
}

// Funzione 3
double Function3::f(double x, double y){
    return (x*x - 1)*(x*x - 1) + y*y;
}
std::array<double,2> Function3::grad(double x, double y){
    return {4*x*(x*x - 1), 2*y};
}

// Funzione 4
double Function4::f(double x, double y){
    return (x*x - 1)*(x*x - 1) + (x - 1)*(x - 1)/5 + y*y;
}
std::array<double,2> Function4::grad(double x, double y){
    return {4*x*(x*x - 1) + 2*(x - 1)/5, 2*y};
}

// Funzione 5
double Function5::f(double x, double y){
    double s = std::sinh(x/2);
    return 4*s*s + y*y;
}
std::array<double,2> Function5::grad(double x, double y){
    return {2*std::sinh(x), 2*y};   // d/dx [4 sinh^2(x/2)] = 2 sinh(x)
}

// Funzione 6
double Function6::f(double x, double y){
    return std::log(1 + x*x + y*y);
}
std::array<double,2> Function6::grad(double x, double y){
    double d = 1 + x*x + y*y;
    return {2*x/d, 2*y/d};
}

// Funzione 7
double Function7::f(double x, double y){
    return 1 - std::exp(-(x*x + y*y));
}
std::array<double,2> Function7::grad(double x, double y){
    double e = std::exp(-(x*x + y*y));
    return {2*x*e, 2*y*e};
}

// Funzione 8
const double PI = std::acos(-1.0);   // M_PI non è standard C++

double Function8::f(double x, double y){
    return 2 - std::cos(PI*x) - std::cos(PI*y);
}
std::array<double,2> Function8::grad(double x, double y){
    return {PI*std::sin(PI*x), PI*std::sin(PI*y)};
}
